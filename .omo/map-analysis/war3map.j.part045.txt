function Jos takes integer jos,integer Kos,code Eos returns nothing
if not HaveSavedHandle(T,jos,Kos) then
call SaveTriggerHandle(T,jos,Kos,CreateTrigger())
endif
call TriggerAddCondition(LoadTriggerHandle(T,jos,Kos),Condition(Eos))
endfunction
function dfs takes trigger mfs returns nothing
local timer Yfs
if wj==mfs then
set Yfs=null
return
endif
set wj=mfs
call TriggerAddAction(mfs,Ax)
if Dj then
set Yfs=CreateTimer()
call SaveTriggerHandle(Sj,0,GetHandleId(Yfs),mfs)
call TimerStart(Yfs,0.,false,Ix)
else
set lj[Nj]=mfs
set Nj=Nj+1
endif
set Yfs=null
endfunction
