function HAv takes nothing returns nothing
local unit cAv=GetEventDamageSource()
local unit kAv=GetTriggerUnit()
local player QAv=GetOwningPlayer(cAv)
local unitstate BAv=UNIT_STATE_MAX_LIFE
local real UAv=GetUnitState(kAv,BAv)
if not qQ then
set qQ=true
call TimerStart(GQ,.25,true,b2)
endif
if iQ!=UAv then
set iQ=UAv
if mQ<UAv then
else
set YQ=YQ*(UAv/mQ)
endif
set mQ=UAv
endif
set YQ=YQ+GetEventDamage()
set BAv=UNIT_STATE_MAX_LIFE
if GetUnitState(kAv,BAv)>=YQ then
if GetPlayerController(QAv)==MAP_CONTROL_USER then
set dQ[GetPlayerId(QAv)]=dQ[GetPlayerId(QAv)]+GetEventDamage()
else
set dQ[GetUnitUserData(cAv)]=dQ[GetUnitUserData(cAv)]+GetEventDamage()
endif
else
set UAv=YQ-UAv
set UAv=GetEventDamage()-UAv
if GetPlayerController(QAv)==MAP_CONTROL_USER then
set dQ[GetPlayerId(QAv)]=dQ[GetPlayerId(QAv)]+UAv
else
set dQ[GetUnitUserData(cAv)]=dQ[GetUnitUserData(cAv)]+UAv
endif
endif
set cAv=null
set kAv=null
set QAv=null
set BAv=null
endfunction
