function Nbs takes nothing returns nothing
local string Wbs=""
local unitstate dbs=UNIT_STATE_MAX_LIFE
local unit Gbs=GetTriggerUnit()
local real ibs=GetUnitState(Gbs,dbs)
local integer Ybs=0
local real mbs
local integer qbs
local integer Zbs
local player ybs
local real lbs
loop
exitwhen Ybs>=4
if as[Ybs]then
call wIs(Ybs)
else
call Yms(Ybs)
endif
set Ybs=Ybs+1
endloop
call PauseTimer(MQ)
set gQ=false
set qbs=0
set Zbs=0
loop
exitwhen qbs>=4
if ks[qbs] and GetPlayerSlotState(Player(qbs))==PLAYER_SLOT_STATE_PLAYING and GetPlayerController(Player(qbs))==MAP_CONTROL_USER then
set Zbs=Zbs+1
endif
set qbs=qbs+1
endloop
if Zbs==2 then
set uQ=30
elseif Zbs==3 then
set uQ=25
else
set uQ=20
endif
set Ybs=0
loop
exitwhen Ybs>=4
if ks[Ybs]then
set mbs=100.*(bQ[Ybs]/ibs)
set nQ[Ybs]=mbs>=I2R(uQ)
set Zbs=R2I(($12c-mR[11])*mbs*5./100.)
call DisplayTimedTextToPlayer(Player(Ybs),0.,0.,5.,"|cffffd700자신이 가한데미지는 "+R2SW(mbs,0,2)+"% 입니다.")
if Zbs>0 then

call Nss(Ybs,Zbs)
endif
if AQ<=bQ[Ybs]then
set AQ=bQ[Ybs]
set IQ=Ybs
set XQ=mbs
endif
endif
set Ybs=Ybs+1
