function Nhs takes string Ghs,string qhs returns nothing
local integer yhs=0
local integer lhs=StringLength(Ghs)
local string Zhs
set YR[0]=""
set dR=0
loop
exitwhen yhs>=lhs
set Zhs=SubString(Ghs,yhs,yhs+1)
if Zhs==qhs then
set dR=dR+1
set YR[dR]=""
else
set YR[dR]=YR[dR]+Zhs
endif
set yhs=yhs+1
endloop
set dR=dR+1
endfunction
