function dSv takes nothing returns nothing
local location pSv=GetUnitLoc(GetEnumUnit())
local unit rSv
local string WSv
local unitstate iSv
local unit mSv
local unit YSv
if GetUnitTurnSpeed(GetEnumUnit())!=.49 then
call SetUnitTurnSpeed(GetEnumUnit(),.49)
set rSv=CreateUnitAtLoc(ds,$68304335,pSv,bj_UNIT_FACING)
call SetUnitScale(rSv,1.4,1.,1.)
call UnitApplyTimedLife(rSv,$42487765,1.)
call SetUnitFlyHeight(rSv,120.,.0)
set WSv="dmlianr.mdx"
if Nv then
set WSv=""
endif
call DestroyEffect(AddSpecialEffectTarget(WSv,rSv,"origin"))
set YSv=LoadUnitHandle(gj,0,Op*$1f4)
set rSv=GetEnumUnit()
set iSv=UNIT_STATE_MAX_LIFE
set mSv=GetEnumUnit()
call yDs(YSv,rSv,GetUnitState(mSv,iSv)*.03,1.,1.,ATTACK_TYPE_NORMAL,DAMAGE_TYPE_UNIVERSAL)
call wVv(GetEnumUnit(),5)
set rSv=GetEnumUnit()
if GetUnitAbilityLevel(rSv,$4130564a)==0 then
call UnitAddAbility(rSv,$4130564a)
endif
call SetUnitAbilityLevel(rSv,$4130564a,GetUnitAbilityLevel(rSv,$4130564a)+1)
call SetUnitVertexColor(GetEnumUnit(),$8a,$8a,$8a,$ff)
endif
call RemoveLocation(pSv)
set pSv=null
set rSv=null
set iSv=null
set mSv=null
set YSv=null
endfunction

function nYs takes integer uYs returns nothing
local integer AYs=GetPlayerId(GetTriggerPlayer())
local integer IYs
local integer XYs
if uYs<=5 then
if GetLocalPlayer()==Player(AYs) then
set IYs=Zi
call BlzFrameSetVisible(LoadFrameHandle(pv,0,IYs),false)
set IYs=ii
call BlzFrameSetVisible(LoadFrameHandle(pv,0,IYs),false)
set IYs=mi
call BlzFrameSetVisible(LoadFrameHandle(pv,0,IYs),false)
set IYs=vi
call BlzFrameSetVisible(LoadFrameHandle(pv,0,IYs),false)
set IYs=Ri
call BlzFrameSetVisible(LoadFrameHandle(pv,0,IYs),false)
set IYs=si
call BlzFrameSetVisible(LoadFrameHandle(pv,0,IYs),true)
set IYs=1
loop
exitwhen IYs>40
set Si[IYs]=LoadInteger(oi,uYs,IYs)
if Si[IYs]==0 then
call rhs(UJ[IYs],false)
else
call lIs(UJ[IYs],LoadStr(ji,0,Si[IYs]))
call qHR(BJ[IYs],I2S(LoadInteger(dv,Fs[AYs],Si[IYs]))+"회")
call rhs(UJ[IYs],true)
endif
set IYs=IYs+1
endloop
endif
elseif uYs<=14 then
if GetLocalPlayer()==Player(AYs) then
set XYs=ii
call BlzFrameSetVisible(LoadFrameHandle(pv,0,XYs),false)
set XYs=mi
call BlzFrameSetVisible(LoadFrameHandle(pv,0,XYs),false)
set XYs=Zi
call BlzFrameSetVisible(LoadFrameHandle(pv,0,XYs),true)

elseif uYs==11 then

elseif uYs==12 then
set Di=Di+7
elseif uYs==13 then
set Di=Di+8
elseif uYs==14 then
set Di=Di+9
endif
set XYs=Vi[Di]
set IYs=1
loop
exitwhen IYs>45
set Si[IYs]=LoadInteger(oi,XYs,IYs)
if Si[IYs]==0 then
call rhs(sJ[IYs],false)
else
call lIs(sJ[IYs],LoadStr(ji,0,Si[IYs]))

endif

set XYs=uYs-14

set IYs=IYs+1

set mVR=GetHandleId(iVR)
call BlzFrameSetEnable(LoadFrameHandle(pv,0,mVR),true)
call nYs(WVR)

set YVR=null
endfunction
